{{- /* Highwire Press metadata for Google Scholar. Scholar reads these tags off the landing page to index it as a paper. CELE's publications are working papers and reports, not journal articles, so they carry the citation_technical_report_* pair rather than citation_journal_title — claiming to be a journal makes Scholar mis-classify the record. Keep at most one tag per field (citation_author is the exception) and keep the values identical to what the PDF's first page says: Scholar cross-checks them and drops records that disagree. */ -}} {{- $pdf := "" -}} {{- with .Params.file -}} {{- $pdf = . | absURL -}} {{- else -}} {{- with .Params.link -}} {{- if strings.HasSuffix (lower .) ".pdf" -}}{{- $pdf = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $report := .Params.reference -}} {{- with .Params.reference_value -}} {{- $report = printf "%s No. %s" $report . -}} {{- end -}} {{- $language := "" -}} {{- with .Params.language -}} {{- if eq . "ESP" -}}{{- $language = "es" -}}{{- else if eq . "ENG" -}}{{- $language = "en" -}}{{- end -}} {{- end -}} {{- range .Params.author }} {{- end }} {{- if not .Date.IsZero }} {{- end }} {{- with $report }} {{- end }} {{- with $pdf }} {{- end }} {{- with $language }} {{- end }} {{- with .Params.doi }} {{- end }} {{- with .Params.issues }} {{- end }}
