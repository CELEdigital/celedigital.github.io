{{- /* Parses one of the Observatorio CSVs into a slice of header→value maps. Call it through partialCached, keyed on the path — the files hold ~2.200 rows between them and several bulletins may embed the shortcode in one build: {{ $rows := partialCached "func/observatorio-data.html" $path $path }} Uses transform.Unmarshal rather than splitting on newlines (the approach in observatory-hub.html), because the Extracto column is full of quoted commas. */ -}} {{- $path := . -}} {{- $rows := slice -}} {{- $raw := readFile $path -}} {{- if not $raw -}} {{- errorf "observatorio-data: no se pudo leer %q" $path -}} {{- end -}} {{- $table := $raw | transform.Unmarshal (dict "delimiter" ",") -}} {{- $header := index $table 0 -}} {{- range $i, $line := $table -}} {{- if gt $i 0 -}} {{- $row := dict -}} {{- range $j, $key := $header -}} {{- $row = merge $row (dict $key (index $line $j)) -}} {{- end -}} {{- $rows = $rows | append $row -}} {{- end -}} {{- end -}} {{- return $rows -}}
